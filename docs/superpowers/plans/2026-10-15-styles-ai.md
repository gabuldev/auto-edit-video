```python
def _png(pixels: dict[tuple[int, int, int, int], int], size=64) -> bytes:
    """An image made of horizontal bands: color → number of rows."""
    img = Image.new("RGBA", (size, sum(pixels.values())))
    y = 0
    for color, rows in pixels.items():
        for yy in range(y, y + rows):
            for x in range(size):
                img.putpixel((x, yy), color)
        y += rows
    buf = io.BytesIO()
    img.save(buf, "PNG")
    return buf.getvalue()


def _close(hex_a: str, hex_b: str, tol=24) -> bool:
    a = [int(hex_a[i:i + 2], 16) for i in (1, 3, 5)]
    b = [int(hex_b[i:i + 2], 16) for i in (1, 3, 5)]
    return all(abs(x - y) <= tol for x, y in zip(a, b))
```

```python
"""The colors of an image (a logo, a reference thumbnail) and a style's
colors from them. Pillow only, no model: an agent never sees the image, it
gets these colors as text, so it works with every agent."""
from __future__ import annotations

import colorsys
import io
from pathlib import Path

from PIL import Image, UnidentifiedImageError


class PaletteError(ValueError):
    pass


def _hex(rgb) -> str:
    return "#{:02x}{:02x}{:02x}".format(*(int(c) for c in rgb[:3]))


def _rgb(hex_color: str) -> tuple[int, int, int]:
    h = hex_color.lstrip("#")
    return tuple(int(h[i:i + 2], 16) for i in (0, 2, 4))  # type: ignore[return-value]


def _hsv(rgb) -> tuple[float, float, float]:
    return colorsys.rgb_to_hsv(*(c / 255 for c in rgb[:3]))


def _open(image: Path | str | bytes) -> Image.Image:
    try:
        img = Image.open(io.BytesIO(image)) if isinstance(image, (bytes, bytearray)) else Image.open(image)
        img = img.convert("RGBA")
    except (UnidentifiedImageError, OSError, ValueError) as exc:
        raise PaletteError("não consegui ler a imagem") from exc
    img.thumbnail((128, 128))
    return img


def extract(image: Path | str | bytes, n: int = 5) -> list[str]:
    """Up to `n` vivid colors, most present first. Transparent, gray,
    near-white and near-black pixels don't count; [] when nothing is left."""
    pixels = [p[:3] for p in _open(image).getdata() if p[3] >= 128]
    vivid = [p for p in pixels if (lambda h, s, v: s >= 0.15 and v >= 0.12)(*_hsv(p))]
    if not vivid:
        return []
    strip = Image.new("RGB", (len(vivid), 1))
    strip.putdata(vivid)
    q = strip.quantize(colors=12, method=Image.Quantize.MEDIANCUT)
    pal = q.getpalette() or []
    scored = []
    for count, idx in q.getcolors() or []:
        rgb = tuple(pal[idx * 3:idx * 3 + 3])
        scored.append((count / len(vivid) * (0.5 + _hsv(rgb)[1]), rgb))
    out: list[tuple[int, int, int]] = []
    for _, rgb in sorted(scored, reverse=True):
        if all(sum(abs(a - b) for a, b in zip(rgb, o)) > 40 for o in out):
            out.append(rgb)
    return [_hex(c) for c in out[:n]]


def _luminance(rgb) -> float:
    def ch(c):
        c /= 255
        return c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4
    r, g, b = (ch(c) for c in rgb)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def _contrast(a, b) -> float:
    la, lb = sorted((_luminance(a), _luminance(b)), reverse=True)
    return (la + 0.05) / (lb + 0.05)


def suggest(colors: list[str]) -> dict | None:
    """A style's colors from a palette: the most saturated as the accent, a
    dark panel of its hue, and the text color that reads best on the panel."""
    if not colors:
        return None
    accent = max(colors, key=lambda c: _hsv(_rgb(c))[1])
    h, _, _ = _hsv(_rgb(accent))
    panel = tuple(round(c * 255) for c in colorsys.hsv_to_rgb(h, 0.6, 0.12))
    text = max(("#ffffff", "#111111"), key=lambda t: _contrast(_rgb(t), panel))
    return {"accent": accent, "panel": _hex(panel), "text": text}
```

```python
def _complete_one(name: str, prompt: str, timeout: float, env: dict) -> str:
    agent = AGENTS[name]
    model = env.get(agent.model_env) or None
    if name in ("claude", "cursor"):
        binary = find_binary(agent)
        if binary is None:
            raise AgentError(f"{agent.label} não encontrado")
    if name == "claude":
        return _run([binary, *(["--model", model] if model else []), "-p"], prompt, timeout, env=env)
    if name == "cursor":
        from auto_edit import runner

        with tempfile.TemporaryDirectory(prefix="auto-edit-agent-") as td:
            prompt_path, out_path = Path(td) / "prompt.txt", Path(td) / "out.txt"
            prompt_path.write_text(prompt, encoding="utf-8")
            rc = runner.invoke_cursor(prompt_path, out_path, Path(__file__).resolve().parent.parent)
            text = out_path.read_text(encoding="utf-8") if out_path.is_file() else ""
            if rc != 0:
                raise AgentError(f"cursor saiu com {rc}: {text.strip()[-300:]}")
            return text
    if name in AGENTIC:
        prompt = TEXT_ONLY + prompt
    return RUNNERS[name](prompt, model=model, timeout=timeout)


def complete(prompt: str, *, timeout: float = 120) -> str:
    """One text answer outside the pipeline (no workspace, no ralph.sh): the
    agent chosen in Settings, then the fallback. AgentError lists what each
    one said when none answers."""
    from auto_edit import settings

    env = settings.env_from_settings()
    names = [normalize(env.get("AUTO_EDIT_LLM") or "claude") or "claude"]
    fallback = normalize(env.get("AUTO_EDIT_LLM_FALLBACK") or "")
    if fallback and fallback not in names:
        names.append(fallback)
    errors = []
    for name in names:
        try:
            return _complete_one(name, prompt, timeout, env)
        except AgentError as exc:
            errors.append(f"{name}: {exc}")
    raise AgentError("; ".join(errors))
```

```python
    dec = json.JSONDecoder()

    """A draft style from a description and/or a reference image (its colors,
    as text). Never saved here; the answer goes through validate()."""
    from auto_edit import agents, palette

    description = (description or "").strip()[:500]
    if not description and not image:
        raise StyleError("descreva o estilo ou mande uma imagem")
    colors: list[str] = []
    if image:
        try:
            colors = palette.extract(image)
        except palette.PaletteError as exc:
            raise StyleError(str(exc)) from exc
    run = complete or (lambda p: agents.complete(p, timeout=AI_TIMEOUT))
    try:
        answer = run(_stylist_prompt(description, colors))
    except agents.AgentError as exc:
        raise StyleError(f"a IA falhou: {exc}") from exc
    data = _first_json(answer)
    if data is None:
        raise StyleError("a IA não devolveu um estilo")
    tokens = data.get("tokens") if isinstance(data.get("tokens"), dict) else data
    name = data.get("name") if isinstance(data.get("name"), str) else ""
    desc = data.get("description") if isinstance(data.get("description"), str) else ""
    return {"name": name.strip()[:40] or "Estilo IA", "description": desc.strip()[:200], "tokens": validate(tokens)}
```

```python
FILE_FORMAT = 1
FILE_SUFFIX = ".autoedit-style.json"
NOT_A_STYLE = "arquivo não é um estilo do Auto-Edit"


def export(style_id: str) -> dict:
    """The shareable file of a style: its look only, never the brand."""
    s = get(style_id)
    if not s:
        raise StyleError(f"estilo não encontrado: {style_id}")
    return {"format": FILE_FORMAT, "name": s["name"], "description": s["description"], "tokens": s["tokens"]}


def export_file(style_id: str, folder: Path) -> Path:
    data = export(style_id)
    folder.mkdir(parents=True, exist_ok=True)
    path, n = folder / f"{style_id}{FILE_SUFFIX}", 2
    while path.exists():
        path, n = folder / f"{style_id}-{n}{FILE_SUFFIX}", n + 1
    path.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
    return path


def _unique_name(name: str) -> str:
    taken = {s["name"] for s in list_styles()}
    out, n = name, 2
    while out in taken:
        out, n = f"{name[:34]} ({n})", n + 1
    return out


def import_(data) -> dict:
    """Save a shared style as a new one of the person's (never overwrites)."""
    if not isinstance(data, dict) or data.get("format") != FILE_FORMAT or not isinstance(data.get("tokens"), dict):
        raise StyleError(NOT_A_STYLE)
    name = str(data.get("name") or "").strip()[:40] or "Estilo importado"
    return save(_unique_name(name), data["tokens"], None, str(data.get("description") or ""))
```

```python
def os_open(path: Path, reveal: bool = False, run: Callable[..., object] = subprocess.Popen) -> None:
    """Open a file with the OS (its default app), or show it selected in the
    file manager (`reveal`)."""
    if sys.platform == "darwin":
        cmd = ["open", "-R", str(path)] if reveal else ["open", str(path)]
    elif sys.platform == "win32":
        cmd = ["explorer", f"/select,{path}"] if reveal else ["cmd", "/c", "start", "", str(path)]
    else:  # no portable "select in folder" on Linux: open the folder instead
        cmd = ["xdg-open", str(path.parent if reveal else path)]
    run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
```

```python
    def _image_bytes(data_url) -> bytes | None:
        """A data: URL from the app → bytes; None when absent. ValueError when wrong."""
        import base64
        import re

        if not data_url:
            return None
        m = re.fullmatch(r"data:image/[a-z+.-]+;base64,([A-Za-z0-9+/=\s]+)", str(data_url).strip())
        if not m:
            raise ValueError("envie uma imagem PNG, JPG ou WebP")
        data = base64.b64decode(m.group(1))
        if len(data) > 5 * 1024 * 1024:
            raise ValueError("imagem grande demais (máx. 5 MB)")
        return data

        except (ValueError, styles.StyleError) as exc:
```

```python
"""`auto-edit style`: create a style with AI, export and import style files."""
from __future__ import annotations

import json
from pathlib import Path
from typing import Optional

import typer
from rich.console import Console

from auto_edit import styles

style_app = typer.Typer(help="Estilos visuais: criar com IA, exportar e importar.", no_args_is_help=True)
console = Console()


def _fail(msg: str) -> None:
    console.print(f"[red]{msg}[/red]")
    raise typer.Exit(1)


@style_app.command("generate")
def generate(
    description: str = typer.Argument("", help="Como o estilo deve ser (ex.: 'tech escuro, neon roxo')"),
    image: Optional[Path] = typer.Option(None, "--image", "-i", help="Imagem de referência (usa as cores dela)"),
    save: bool = typer.Option(False, "--save", help="Salva como um estilo seu"),
    activate: bool = typer.Option(False, "--activate", help="Salva e passa a usar"),
):
    """Cria um estilo com a IA (rascunho; --save grava)."""
    try:
        draft = styles.generate(description, image.read_bytes() if image else None)
    except (OSError, styles.StyleError) as exc:
        _fail(str(exc))
    if save or activate:
        saved = styles.save(draft["name"], draft["tokens"], None, draft["description"])
        if activate:
            styles.set_active(saved["id"])
        console.print(f"[green]salvo:[/green] {saved['name']} ({saved['id']})")
    else:
        typer.echo(json.dumps(draft, indent=2, ensure_ascii=False))


@style_app.command("export")
def export(
    style_id: str = typer.Argument(..., help="Id do estilo"),
    output: Optional[Path] = typer.Option(None, "--output", "-o", help=f"Arquivo (padrão: <id>{styles.FILE_SUFFIX})"),
):
    """Exporta um estilo pra um arquivo compartilhável (sem a sua marca)."""
    try:
        data = styles.export(style_id)
    except styles.StyleError as exc:
        _fail(str(exc))
    out = output or Path(f"{style_id}{styles.FILE_SUFFIX}")
    out.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
    console.print(f"[green]exportado:[/green] {out}")


@style_app.command("import")
def import_(
    file: Path = typer.Argument(..., help="Arquivo .autoedit-style.json"),
    activate: bool = typer.Option(False, "--activate", help="Passa a usar"),
):
    """Importa um estilo (vira um estilo seu, nunca sobrescreve)."""
    try:
        data = json.loads(file.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        _fail(styles.NOT_A_STYLE)
    try:
        saved = styles.import_(data)
    except styles.StyleError as exc:
        _fail(str(exc))
    if activate:
        styles.set_active(saved["id"])
    console.print(f"[green]importado:[/green] {saved['name']} ({saved['id']})")
```
